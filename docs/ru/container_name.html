Контейнер

Реализация контейнера в VyOS основана на Podman как движок без сервера.

Конфигурация

set container name <name> image

Устанавливает имя образа в реестре хаба

set container name mysql-server image mysql:8.0

Если реестр не указан, будет использоваться Docker.io как реестр контейнера, если только альтернативный реестр не указан с помощью set container registry <name> или реестр указан в имени образа

set container name mysql-server image quay.io/mysql:8.0
set container name <name> entrypoint <entrypoint>

Переопределяет стандартный точку входа из образа для контейнера.

set container name <name> command <command>

Переопределяет стандартную команду из образа для контейнера.

set container name <name> arguments <arguments>

Устанавливает аргументы команды для контейнера.

set container name <name> host-name <hostname>

Устанавливает имя хоста для контейнера.

set container name <name> allow-host-pid

Контейнер и хост используют одинаковое пространство имен процессов. Это означает, что процессы, запущенные на хосте, видны внутри контейнера, и процессы внутри контейнера видны на хосте.

Команда преобразуется в «–pid host» при создании контейнера.

set container name <name> allow-host-networks

Позволяет использовать сетевой стек хоста в контейнере. Сетевой стек контейнера не изолирован от хоста и будет использовать IP хоста.

Команда преобразуется в «–net host» при создании контейнера.

Примечание

allow-host-networks не может быть использовано с network

set container name <name> network <networkname>

Присоединяет пользовательскую сеть к контейнеру. Должно быть указано только одно сетевое имя и оно должно уже существовать.

set container name <name> network <networkname> address <address>

Опционально задает конкретный статический IPv4 или IPv6 адрес для контейнера. Этот адрес должен находиться в пределе названного сетевого префикса.

Примечание

Первый IP в сети контейнера зарезервирован двигателем и не может быть использован

set container name <name> name-server <address>

Опционально задает пользовательский сервер имен. Если используется сеть контейнера с включенным DNS, это настройка не будет иметь никакого эффекта.

set container name <name> description <text>

Установить описание контейнера

set container name <name> environment <key> value <value>

Добавляет пользовательские переменные окружения. Разрешено несколько переменных окружения. Следующие команды преобразуются в «-e key=value» при создании контейнера.

set container name mysql-server environment MYSQL_DATABASE value 'zabbix'
set container name mysql-server environment MYSQL_USER value 'zabbix'
set container name mysql-server environment MYSQL_PASSWORD value 'zabbix_pwd'
set container name mysql-server environment MYSQL_ROOT_PASSWORD value 'root_pwd'
set container name <name> port <portname> source <portnumber>
set container name <name> port <portname> destination <portnumber>
set container name <name> port <portname> protocol <tcp | udp>

Публикует порт для контейнера.

set container name zabbix-web-nginx-mysql port http source 80
set container name zabbix-web-nginx-mysql port http destination 8080
set container name zabbix-web-nginx-mysql port http protocol tcp

Примечание

Публикация портов не может быть использована с network. Для этого цели доступен обходной путь с использованием NAT назначения и статического назначения IP для контейнера.

set container name <name> volume <volumename> source <path>
set container name <name> volume <volumename> destination <path>

Монтирует объем в контейнер

set container name coredns volume 'corefile' source /config/coredns/Corefile
set container name coredns volume 'corefile' destination /etc/Corefile
set container name <name> volume <volumename> mode <ro | rw>

Объем монтируется как rw (чтение/запись - по умолчанию) или ro (только чтение)

set container name <name> tmpfs <tmpfsname> destination <path>

Монтирует файловую систему tmpfs (оперативная память) по указанному пути внутри контейнера.

set container name <name> tmpfs <tmpfsname> size <MB>

Размер в МБ для файловой системы tmpfs, максимальный размер 64 ГБ или 50% общего доступного объема памяти системы.

set container name <name> uid <number>
set container name <name> gid <number>

Установите идентификатор пользователя или группы контейнера

set container name <name> restart [no | on-failure | always]

Установите поведение перезапуска контейнера.

  • no: Не перезапускать контейнеры при выходе

  • on-failure: Перезапускать контейнеры при выходе с ненулевым кодом завершения, повторяя бесконечно (по умолчанию)

  • always: Перезапускать контейнеры при выходе, независимо от состояния, повторяя бесконечно

set container name <name> cpu-quota <num>

Это указывает количество ресурсов ЦП, которые может использовать контейнер.

По умолчанию 0 для неограниченного. Например, 1.25 ограничивает контейнер до использования до 1.25 ядер ЦП. Это может быть десятичное число с тремя десятичными знаками.

Команда преобразуется в «–cpus=<num>» при создании контейнера.

set container name <name> memory <MB>

Ограничивает память, доступную для контейнера.

По умолчанию 512 МБ. Используйте 0 МБ для неограниченной памяти.

set container name <name> device <devicename> source <path>
set container name <name> device <devicename> destination <path>

Добавьте устройство хоста в контейнер.

set container name <name> capability <text>

Установите возможности или разрешения контейнера.

  • net-admin: Операции сети (интерфейс, фаерволл, таблицы маршрутизации)

  • net-bind-service: Привязка сокета к привилегированным портам (числа портов меньше 1024)

  • net-raw: Разрешение на создание сырых сетевых сокетов

  • setpcap: Наборы возможностей (из ограниченного или унаследованного набора)

  • sys-admin: Операции администрирования (quotactl, mount, sethostname, setdomainame)

  • sys-time: Разрешение на установку системного часов

set container name <name> sysctl parameter <parameter> value <value>

Установите значения sysctl для контейнера.

Подмножество возможных параметров:

  • Параметры ядра: kernel.msgmax, kernel.msgmnb, kernel.msgmni, kernel.sem, kernel.shmall, kernel.shmmax, kernel.shmmni, kernel.shm_rmid_forced

  • Параметры, начинающиеся с fs.mqueue.*

  • Параметры, начинающиеся с net.* (только если используется пользовательская сеть)

set container name <name> label <label> value <value>

Добавьте метаданные для этого контейнера.

set container name <name> disable

Отключить контейнер.

Проверки состояния контейнера

По умолчанию проверки состояния не выполняются, даже если они определены образом.

set container name <name> health-check

По умолчанию выполняется проверка состояния контейнера, если она определена образом.

set container name <name> health-check command <command>

Переопределяет команду проверки состояния по умолчанию из образа для контейнера.

set container name <name> health-check interval <interval>

Переопределяет интервал проверки состояния по умолчанию. Например: 60

set container name <name> health-check timeout <timeout>

Переопределяет время ожидания проверки состояния по умолчанию. Например: 10

set container name <name> health-check retries <retries>

Количество попыток проверки состояния перед тем, как контейнер считается неправильным. Например: 1

Сети контейнеров

set container network <name>

Создает сетевое имя контейнера

set container network <name> description

Краткое описание того, о чем эта сеть.

set container network <name> prefix <ipv4|ipv6>

Определяет префикс IPv4 и/или IPv6 для заданного имени сети. IPv4 и IPv6 могут использоваться параллельно.

set container network <name> mtu <number>

Настройка MTU для заданной сети. Это размер (в байтах) самого большого фрейма Ethernet, отправляемого на этом соединении.

set container network <name> no-name-server

Отключает плагин системы доменных имен (DNS) для этой сети.

set container network <name> vrf <name>

Привязывает сеть контейнера к заданному экземпляру VRF.

Реестр контейнеров

set container registry <name>

Добавляет реестр в список unqualified-search-registries. По умолчанию, для любого образа, который не включает реестр в имя образа, VyOS будет использовать docker.io и quay.io как реестр контейнера.

set container registry <name> disable

Отключить заданный реестр контейнера

set container registry <name> authentication username
set container registry <name> authentication password

Некоторые реестры контейнеров требуют учетных данных для использования.

Учетные данные можно определить здесь и будут использоваться только при добавлении образа контейнера в систему.

set container registry <name> insecure

Разрешает доступ к реестру через незашифрованный HTTP или TLS-соединения с непроверенными сертификатами.

set container registry <name> mirror address <address>
set container registry <name> mirror host-name <host-name>
set container registry <name> mirror port <port>
set container registry <name> mirror path <path>

Зеркало реестра, используйте (host-name|address)[:port][/path].

Если у вас есть зеркало http://192.168.1.1:8080 для docker.io, вы можете использовать docker.io/some/repo или выполнить podman pull docker.io/some/repo

set container registry docker.io mirror address 192.168.1.1
set container registry docker.io mirror port 8080
set container registry docker.io insecure

Если http://192.168.1.1:8080 является вашим собственным реестром, вы можете использовать 192.168.1.1:8080/some/repo или выполнить podman pull 192.168.1.1:8080/some/repo

set container registry 192.168.1.1:8080 insecure

Конфигурация журнала

set container name <name> log-driver [k8s-file | journald | none]

Установите драйвер журнала по умолчанию для контейнеров.

  • k8s-file: Журналирование в текстовый файл в стиле Kubernetes.

  • journald: Журналирование в системный журнал

  • none: Отключить журналирование для контейнера

Текущее значение по умолчанию — journald.

Команды операций

add container image <containername>

Загрузить новый образ для контейнера

show container

Показать список всех активных контейнеров.

show container image

Показать локальные образы контейнеров.

show container log <containername>

Показать журналы из заданного контейнера

show container network

Показать список доступных сетей контейнеров

restart container <containername>

Перезапустить заданный контейнер

update container image <containername>

Обновить образ контейнера

delete container image <image id|all> [force]

Удалить определенный образ контейнера на основе его идентификатора образа. Вы также можете удалить все образы контейнеров одновременно.

Вы не можете удалить образ контейнера, если у него более одного тега, присвоенного, поэтому есть опция force, чтобы передать ее в образ контейнера, чтобы также удалить эти образы.

Пример конфигурации

В целях демонстрации, пример #1 в официальной документации к декларативному синтаксису CLI VyOS.

set container network zabbix prefix 172.20.0.0/16
set container network zabbix description 'Сеть для компонентов контейнеров Zabbix'

set container name mysql-server image mysql:8.0
set container name mysql-server network zabbix

set container name mysql-server environment 'MYSQL_DATABASE' value 'zabbix'
set container name mysql-server environment 'MYSQL_USER' value 'zabbix'
set container name mysql-server environment 'MYSQL_PASSWORD' value 'zabbix_pwd'
set container name mysql-server environment 'MYSQL_ROOT_PASSWORD' value 'root_pwd'

set container name zabbix-java-gateway image zabbix/zabbix-java-gateway:alpine-5.2-latest
set container name zabbix-java-gateway network zabbix

set container name zabbix-server-mysql image zabbix/zabbix-server-mysql:alpine-5.2-latest
set container name zabbix-server-mysql network zabbix

set container name zabbix-server-mysql environment 'DB_SERVER_HOST' value 'mysql-server'
set container name zabbix-server-mysql environment 'MYSQL_DATABASE' value 'zabbix'
set container name zabbix-server-mysql environment 'MYSQL_USER' value 'zabbix'
set container name zabbix-server-mysql environment 'MYSQL_PASSWORD' value 'zabbix_pwd'
set container name zabbix-server-mysql environment 'MYSQL_ROOT_PASSWORD' value 'root_pwd'
set container name zabbix-server-mysql environment 'ZBX_JAVAGATEWAY' value 'zabbix-java-gateway'

set container name zabbix-server-mysql port zabbix source 10051
set container name zabbix-server-mysql port zabbix destination 10051

set container name zabbix-web-nginx-mysql image zabbix/zabbix-web-nginx-mysql:alpine-5.2-latest
set container name zabbix-web-nginx-mysql network zabbix

set container name zabbix-web-nginx-mysql environment 'MYSQL_DATABASE' value 'zabbix'
set container name zabbix-web-nginx-mysql environment 'ZBX_SERVER_HOST' value 'zabbix-server-mysql'
set container name zabbix-web-nginx-mysql environment 'DB_SERVER_HOST' value 'mysql-server'
set container name zabbix-web-nginx-mysql environment 'MYSQL_USER' value 'zabbix'
set container name zabbix-web-nginx-mysql environment 'MYSQL_PASSWORD' value 'zabbix_pwd'
set container name zabbix-web-nginx-mysql environment 'MYSQL_ROOT_PASSWORD' value 'root_pwd'

set container name zabbix-web-nginx-mysql port http source 80
set container name zabbix-web-nginx-mysql port http destination 8080
